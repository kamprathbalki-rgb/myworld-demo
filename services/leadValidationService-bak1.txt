exports.validateMobile = (mobile) => {

    if (!mobile) {
        return {
            valid: false,
            reason: "missing"
        };
    }

    const cleaned = mobile.replace(/\D/g, "");

    if (cleaned.length !== 10) {
        return {
            valid: false,
            reason: "invalid"
        };
    }

    return {
        valid: true,
        value: cleaned
    };

};

exports.validateEmail = (email) => {

    if (!email) {
        return {
            valid: false,
            reason: "missing"
        };
    }

    const valid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);

    if (!valid) {
        return {
            valid: false,
            reason: "invalid"
        };
    }

    return {
        valid: true,
        value: email.trim().toLowerCase()
    };

};

exports.needsVerification = (lead) => {

    return (
        lead.mobile &&
        lead.email &&
        (
            !lead.mobileVerified ||
            !lead.emailVerified
        )
    );

};

exports.missingFields = (lead) => {

    const missing = [];

if (!lead.propertyCategory)
    missing.push("propertyCategory");

if (!lead.location)
    missing.push("location");

if (!lead.propertyType)
    missing.push("propertyType");

if (!lead.configuration)
    missing.push("configuration");

if (!lead.budget)
    missing.push("budget");

if (!lead.name && !lead.nameDeclined)
    missing.push("name");

if (!lead.mobile && !lead.mobileDeclined)
    missing.push("mobile");

if (!lead.email && !lead.emailDeclined)
    missing.push("email");

    return missing;

};

exports.nextQuestion = (missing, lead = {}) => {

    if (!missing.length)
        return null;

    const asked = new Set(
        Object.keys(lead)
            .filter(k => lead[k] !== undefined && lead[k] !== null && lead[k] !== "")
    );

    if (
        asked.has("location") &&
        asked.has("propertyType") &&
        asked.has("configuration") &&
        !asked.has("budget")
    ) {
        return "What's your approximate budget?";
    }

    if (
        asked.has("budget") &&
        !asked.has("name")
    ) {
        return "Before I suggest suitable properties, how would you like me to address you?";
    }

    if (
        asked.has("name") &&
        !asked.has("mobile")
    ) {
        return `Thanks ${lead.name}. If you're comfortable, you can share your mobile number for updates. Otherwise we can continue here.`;
    }

    if (
        asked.has("name") &&
        !asked.has("email")
    ) {
        return "Would you like to share your email so I can send brochures? It's optional.";
    }

    const questions = {

        propertyCategory:
            "Are you looking for a residential, commercial or plot property?",

        location:
            "Which city or location are you interested in?",

    propertyType:
    lead.propertyCategory === "Residential"
        ? "What type of residential property are you looking for? (Apartment, Villa, Row House, Independent House, Penthouse)"
        : lead.propertyCategory === "Commercial"
        ? "What type of commercial property are you looking for? (Office, Shop, Showroom, Warehouse)"
        : "What type of plot are you looking for? (Residential Plot, Commercial Plot, Agricultural Land, Industrial Land)",

        configuration:
            "What configuration are you looking for (1 BHK, 2 BHK, Villa, Plot etc.)?",

        budget:
            "What's your approximate budget?",

        name:
            "May I know your name?",

        mobile:
            "If you're comfortable, could you share your mobile number? We can continue even if you don't.",

        email:
            "Would you like to share your email address? It's optional."

    };

    return questions[missing[0]];

};
