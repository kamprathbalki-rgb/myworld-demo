            card.dataset.matches;

            card.dataset.possession;

        document.getElementById("drawerPreferred").textContent =
            card.dataset.preferred;

        document.getElementById("drawerVisit").textContent =
            card.dataset.visit;

            card.dataset.activity;

        document.getElementById("drawerDeleteForm").action =
            "/buyer/delete/" + card.dataset.id;

    "🏠 " + card.dataset.matches + " Matches";

        drawer.classList.add("open");

    drawer.classList.remove("open");

    if(e.target===drawer){

        drawer.classList.remove("open");

    }
